using TrueMoon.Aether.Runtime;
using TrueMoon.Aether.Scenes;
using TrueMoon.Aether.Sessions;
using TrueMoon.Services;
namespace Headless.Game;
public sealed class Game : IGameSession, IGameSessionEntry
{
    public ulong TickNumber { get; private set; }
    public long Revision { get; private set; }
    public uint PatternColor => 0;
    public static IGameSession Create(IServiceResolver? services) => new Game();
    public void Load(SceneSnapshot scene) { Revision = scene.Revision; TickNumber = 0; }
    public void Tick() { TickNumber++; }
    public void Dispose() { }
}
